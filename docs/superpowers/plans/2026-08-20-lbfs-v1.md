**Goal:** Build the lbfs v1 network filesystem: a Rust FUSE client that proxies VFS operations over a pipelined binary TCP protocol to an io_uring-backed server, plus a libvirt VM test environment.

**Architecture:** Cargo workspace with three crates. `lbfs-proto` owns the wire contract (24-byte frame header, postcard bodies, out-of-band bulk data). `lbfs-server` splits into an `rpc` module (tokio sessions, window enforcement, dispatch) and an `fs` module (async `FileSystem` trait; `LocalFs` executes data-plane ops on a dedicated io_uring thread and the rest via `spawn_blocking`, with virtiofsd-style `O_PATH` node tables and `openat2(RESOLVE_BENEATH)`). `lbfs-client` bridges fuser callbacks onto a tokio multiplexer with a correlation table and window semaphore.

**Tech Stack:** Rust (edition 2021), tokio 1, fuser 0.15, io-uring 0.7, rustix 1, postcard 1.1 + serde/serde_bytes, globset, clap 4, toml, tracing, proptest, tempfile; libvirt + Ubuntu 26.04 cloud images for VMs.

**Spec:** `docs/superpowers/specs/2026-08-20-lbfs-design.md`

## Global Constraints

- Frame header: exactly 24 bytes, little-endian, layout per spec §3.1.
- Protocol magic `LBFS`, version `1`, exact version match required.
- Defaults: port `9423`, window `128` (clamp 8..=1024), max I/O `1 MiB`, max body `64 KiB`.
- Status field: `0` OK, `1..=4095` Linux errno, `>= 0xFF00` protocol statuses.
- Flags: bit 0 `NO_REPLY`; bit 1 reserved for `FORCE_SYNC` — define the constant, never set it.
- Names, symlink targets, xattr names/values are byte strings — never `String`.
- Bulk data never passes through postcard; send with vectored writes, receive into pooled buffers.
- The server never follows symlinks; every path step uses `openat2` with `RESOLVE_BENEATH | RESOLVE_NO_MAGICLINKS`.
- Every task ends green: `make check` (fmt --check, clippy `-D warnings`, tests) passes before every commit.
- TDD: write the failing test first for every behavior.
- No `unsafe` outside `crates/lbfs-server/src/fs/local/uring.rs`.
- Commit after every task with the exact paths staged (no blanket `git add .`).

```rust
pub const HEADER_LEN: usize = 24;
pub const MAGIC: [u8; 4] = *b"LBFS";
pub const PROTOCOL_VERSION: u32 = 1;
pub const DEFAULT_PORT: u16 = 9423;
pub const DEFAULT_MAX_INFLIGHT: u32 = 128;
pub const WINDOW_CLAMP: (u32, u32) = (8, 1024);
pub const DEFAULT_MAX_IO_SIZE: u32 = 1 << 20;
pub const MAX_BODY_SIZE: u32 = 64 << 10;

pub const FLAG_NO_REPLY: u16 = 1 << 0;
/// Reserved for the forced-sync fast-follow (spec §11). Never set in v1.
pub const FLAG_FORCE_SYNC_RESERVED: u16 = 1 << 1;

pub const STATUS_OK: u16 = 0;
pub const STATUS_VERSION_MISMATCH: u16 = 0xFF01;
pub const STATUS_ATTACH_DENIED: u16 = 0xFF02;
pub const STATUS_NOT_EXPORTED: u16 = 0xFF03;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct FrameHeader {
    pub request_id: u64,
    pub op_or_status: u16,
    pub flags: u16,
    pub body_len: u32,
    pub data_len: u32,
}

impl FrameHeader {
    pub fn encode(&self) -> [u8; HEADER_LEN] {
        let mut b = [0u8; HEADER_LEN];
        b[0..8].copy_from_slice(&self.request_id.to_le_bytes());
        b[8..10].copy_from_slice(&self.op_or_status.to_le_bytes());
        b[10..12].copy_from_slice(&self.flags.to_le_bytes());
        b[12..16].copy_from_slice(&self.body_len.to_le_bytes());
        b[16..20].copy_from_slice(&self.data_len.to_le_bytes());
        // b[20..24] reserved, zero
        b
    }

    pub fn decode(b: &[u8; HEADER_LEN]) -> Self {
        Self {
            request_id: u64::from_le_bytes(b[0..8].try_into().unwrap()),
            op_or_status: u16::from_le_bytes(b[8..10].try_into().unwrap()),
            flags: u16::from_le_bytes(b[10..12].try_into().unwrap()),
            body_len: u32::from_le_bytes(b[12..16].try_into().unwrap()),
            data_len: u32::from_le_bytes(b[16..20].try_into().unwrap()),
        }
    }
}
```

```rust
    pub const EIO: Errno = Errno(libc::EIO as u16);
    pub const ESTALE: Errno = Errno(libc::ESTALE as u16);
    pub const ENOSYS: Errno = Errno(libc::ENOSYS as u16);
    pub const EACCES: Errno = Errno(libc::EACCES as u16);
    pub const EINVAL: Errno = Errno(libc::EINVAL as u16);
    pub const ENODATA: Errno = Errno(libc::ENODATA as u16);
    pub const ERANGE: Errno = Errno(libc::ERANGE as u16);
    pub const ENOTEMPTY: Errno = Errno(libc::ENOTEMPTY as u16);
    pub const EEXIST: Errno = Errno(libc::EEXIST as u16);

    pub fn from_io(e: &std::io::Error) -> Errno {
```

```rust
use crate::frame::{FrameHeader, HEADER_LEN};
use std::io::IoSlice;
use tokio::io::{AsyncRead, AsyncReadExt, AsyncWrite, AsyncWriteExt};

#[derive(Debug, thiserror::Error)]
pub enum IoError {
    #[error("io: {0}")]
    Io(#[from] std::io::Error),
    #[error("protocol violation: {0}")]
    Protocol(&'static str),
}

pub async fn read_header<R: AsyncRead + Unpin>(r: &mut R) -> std::io::Result<FrameHeader> {
    let mut buf = [0u8; HEADER_LEN];
    r.read_exact(&mut buf).await?;
    Ok(FrameHeader::decode(&buf))
}

pub async fn read_body<R: AsyncRead + Unpin>(r: &mut R, len: u32, max: u32) -> Result<Vec<u8>, IoError> {
    if len > max {
        return Err(IoError::Protocol("body_len exceeds negotiated maximum"));
    }
    let mut body = vec![0u8; len as usize];
    r.read_exact(&mut body).await?;
    Ok(body)
}

pub async fn write_frame<W: AsyncWrite + Unpin>(
    w: &mut W,
    hdr: FrameHeader,
    body: &[u8],
    data: &[u8],
) -> std::io::Result<()> {
    debug_assert_eq!(hdr.body_len as usize, body.len());
    debug_assert_eq!(hdr.data_len as usize, data.len());
    let head = hdr.encode();
    let mut slices = [IoSlice::new(&head), IoSlice::new(body), IoSlice::new(data)];
    let mut total = HEADER_LEN + body.len() + data.len();
    let mut cursor = 0usize;
    // write_vectored can return short counts: advance across the slice list.
    while total > 0 {
        let n = w.write_vectored(&slices[..]).await?;
        if n == 0 {
            return Err(std::io::ErrorKind::WriteZero.into());
        }
        total -= n;
        cursor += n;
        if total > 0 {
            // Rebuild the slice window past `cursor` bytes.
            let (h, b) = (head.len(), body.len());
            let (hs, bs, ds) = (
                &head[cursor.min(h)..],
                &body[cursor.saturating_sub(h).min(b)..],
                &data[cursor.saturating_sub(h + b)..],
            );
            slices = [IoSlice::new(hs), IoSlice::new(bs), IoSlice::new(ds)];
        }
    }
    w.flush().await
}
```

```rust
use lbfs_proto::frame::{DEFAULT_MAX_INFLIGHT, DEFAULT_MAX_IO_SIZE, WINDOW_CLAMP};
use serde::Deserialize;
use std::path::{Path, PathBuf};

#[derive(Debug, thiserror::Error)]
pub enum ConfigError {
    #[error("toml: {0}")]
    Toml(#[from] toml::de::Error),
    #[error("bad size literal: {0}")]
    BadSize(String),
    #[error("bad glob: {0}")]
    BadGlob(#[from] globset::Error),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, Deserialize)]
#[serde(rename_all = "lowercase")]
pub enum FsyncPolicy { Honor, Ignore }

#[derive(Debug, Deserialize)]
struct RawConfig {
    listen: String,
    allowed_paths: Vec<String>,
    max_inflight: Option<u32>,
    max_io_size: Option<String>,
    fsync: Option<FsyncPolicy>,
}

#[derive(Debug)]
pub struct Config {
    pub listen: String,
    pub allowed_paths: Vec<String>,
    pub max_inflight: u32,
    pub max_io_size: u32,
    pub fsync: FsyncPolicy,
}

impl Config {
    pub fn from_toml(s: &str) -> Result<Config, ConfigError> {
        let raw: RawConfig = toml::from_str(s)?;
        let max_io_size = match raw.max_io_size {
            Some(ref lit) => parse_size(lit)?,
            None => DEFAULT_MAX_IO_SIZE,
        };
        Ok(Config {
            listen: raw.listen,
            allowed_paths: raw.allowed_paths,
            max_inflight: raw
                .max_inflight
                .unwrap_or(DEFAULT_MAX_INFLIGHT)
                .clamp(WINDOW_CLAMP.0, WINDOW_CLAMP.1),
            max_io_size,
            fsync: raw.fsync.unwrap_or(FsyncPolicy::Honor),
        })
    }
}

pub fn parse_size(s: &str) -> Result<u32, ConfigError> {
    let bad = || ConfigError::BadSize(s.to_string());
    if let Some(n) = s.strip_suffix("MiB") {
        return n.trim().parse::<u32>().map(|v| v << 20).map_err(|_| bad());
    }
    if let Some(n) = s.strip_suffix("KiB") {
        return n.trim().parse::<u32>().map(|v| v << 10).map_err(|_| bad());
    }
    s.trim().parse::<u32>().map_err(|_| bad())
}

#[derive(Debug)]
pub enum AttachError { NotExported, Denied }

pub struct Allowlist {
    set: globset::GlobSet,
}

impl Allowlist {
    pub fn new(patterns: &[String]) -> Result<Allowlist, ConfigError> {
        let mut b = globset::GlobSetBuilder::new();
        for p in patterns {
            b.add(globset::GlobBuilder::new(p).literal_separator(true).build()?);
        }
        Ok(Allowlist { set: b.build()? })
    }

    /// Canonicalize FIRST, then match — symlinks cannot smuggle a path
    /// past the allowlist (spec §4).
    pub fn check(&self, requested: &Path) -> Result<PathBuf, AttachError> {
        let canon = std::fs::canonicalize(requested).map_err(|_| AttachError::NotExported)?;
        if self.set.is_match(&canon) {
            Ok(canon)
        } else {
            Err(AttachError::Denied)
        }
    }
}
```

```rust
use std::sync::{Arc, Mutex};

struct Inner {
    bufs: Mutex<Vec<Box<[u8]>>>,
    buf_size: usize,
    max_pooled: usize,
}

#[derive(Clone)]
pub struct BufferPool(Arc<Inner>);

impl BufferPool {
    pub fn new(buf_size: usize, max_pooled: usize) -> BufferPool {
        BufferPool(Arc::new(Inner { bufs: Mutex::new(Vec::new()), buf_size, max_pooled }))
    }

    pub fn get(&self) -> PooledBuf {
        let buf = self.0.bufs.lock().unwrap().pop()
            .unwrap_or_else(|| vec![0u8; self.0.buf_size].into_boxed_slice());
        PooledBuf { buf: Some(buf), len: 0, pool: Arc::downgrade(&self.0) }
    }

    #[cfg(test)]
    pub fn pooled_count(&self) -> usize {
        self.0.bufs.lock().unwrap().len()
    }
}

pub struct PooledBuf {
    buf: Option<Box<[u8]>>,
    len: usize,
    pool: std::sync::Weak<Inner>,
}

impl PooledBuf {
    pub fn as_slice(&self) -> &[u8] { &self.buf.as_ref().unwrap()[..self.len] }
    pub fn as_mut_slice(&mut self) -> &mut [u8] { self.buf.as_mut().unwrap() }
    pub fn set_len(&mut self, n: usize) {
        assert!(n <= self.capacity());
        self.len = n;
    }
    pub fn capacity(&self) -> usize { self.buf.as_ref().unwrap().len() }
}

impl Drop for PooledBuf {
    fn drop(&mut self) {
        if let (Some(buf), Some(inner)) = (self.buf.take(), self.pool.upgrade()) {
            let mut bufs = inner.bufs.lock().unwrap();
            if bufs.len() < inner.max_pooled {
                bufs.push(buf);
            }
        }
    }
}
```

Architecture (one ring thread; `threads` parameter spawns N independent ring threads round-robin'd by an `AtomicUsize`):

```rust
// Submission side (async): build an OpPayload that OWNS every pointer target
// (CString names, PooledBuf, Box<libc::statx>, Box<OpenHow>), push it with a
// oneshot::Sender into an std::sync::mpsc channel, then write 1 to an eventfd
// to wake the ring thread.
//
// Ring thread loop:
//   1. drain the channel: for each task, insert into a slab -> key,
//      build the SQE with user_data = key + 1, push to the SQ
//      (if the SQ is full, submit() first; if the slab is at capacity,
//      park the task in a VecDeque backlog and retry after completions)
//   2. always keep one Read SQE armed on the eventfd with user_data = 0
//   3. submit_and_wait(1)
//   4. for each CQE: user_data 0 => re-arm eventfd read; else remove slab
//      entry, send (cqe.result(), payload) through the oneshot
//
// Completion decode: cqe.result() < 0 => io::Error::from_raw_os_error(-res),
// else success value (bytes for read/write, new fd for openat2).
```

Key types:

```rust
enum OpPayload {
    Rw { buf: PooledBuf },
    XattrRw { buf: PooledBuf, name: CString },
    Path { name: CString },
    TwoPath { a: CString, b: CString },
    Open { name: CString, how: Box<io_uring::types::OpenHow> },
    Statx { name: CString, out: Box<libc::statx> },
    Bare,
}

struct Task {
    op: OpDesc, // enum mirroring the public methods, carrying fds/offsets/lens
    payload: OpPayload,
    done: tokio::sync::oneshot::Sender<(i32, OpPayload)>,
}
```

SQE construction per op (io-uring crate opcodes): `opcode::Read::new(types::Fd(fd), ptr, len).offset(off)`, `opcode::Write`, `opcode::Fsync` (`FsyncFlags::DATASYNC` when datasync), `opcode::Fallocate`, `opcode::OpenAt2::new(types::Fd(dirfd), name_ptr, how_ptr)`, `opcode::Statx::new(types::Fd(dirfd), name_ptr, out_ptr).flags(flags).mask(mask)`, `opcode::UnlinkAt` (`.flags(libc::AT_REMOVEDIR)` for rmdir), `opcode::MkDirAt`, `opcode::RenameAt` (`.flags(flags)` maps `RENAME_NOREPLACE`/`RENAME_EXCHANGE`), `opcode::LinkAt`, `opcode::SymlinkAt`, `opcode::FGetXattr`, `opcode::FSetXattr`. Pointers come from the `OpPayload` the slab owns — they stay valid until the CQE arrives, which is the whole point of the slab.

Public async methods share one helper:

```rust
async fn submit(&self, op: OpDesc, payload: OpPayload) -> (i32, OpPayload) {
    let (tx, rx) = tokio::sync::oneshot::channel();
    let lane = self.next.fetch_add(1, Ordering::Relaxed) % self.lanes.len();
    self.lanes[lane].queue.send(Task { op, payload, done: tx }).expect("ring thread alive");
    self.lanes[lane].wake(); // write(eventfd, 1)
    rx.await.expect("ring thread never drops tasks")
}
```

then e.g.:

```rust
pub async fn read(&self, fd: RawFd, offset: u64, buf: PooledBuf, len: u32)
    -> (PooledBuf, std::io::Result<u32>)
{
    let (res, payload) = self
        .submit(OpDesc::Read { fd, offset, len }, OpPayload::Rw { buf })
        .await;
    let OpPayload::Rw { mut buf } = payload else { unreachable!() };
    if res < 0 {
        (buf, Err(std::io::Error::from_raw_os_error(-res)))
    } else {
        buf.set_len(res as usize);
        (buf, Ok(res as u32))
    }
}
```

Follow the identical decode pattern for every other method (openat2 wraps the
positive result via `OwnedFd::from_raw_fd`; statx moves `out` back by value).
Shutdown: `UringExecutor`'s inner `Arc` drop closes the channel; the ring
thread exits after draining in-flight CQEs.

```rust
use lbfs_proto::types::{NodeId, ROOT_NODE};
use std::collections::HashMap;
use std::os::fd::OwnedFd;
use std::sync::{Arc, Mutex};

pub type FileKey = (u64, u64); // (st_dev, st_ino)

struct Node {
    fd: Arc<OwnedFd>,
    key: FileKey,
    generation: u64,
    nlookup: u64,
}

struct Inner {
    nodes: HashMap<NodeId, Node>,
    by_key: HashMap<FileKey, NodeId>,
    next_id: NodeId,
    next_generation: u64,
}

pub struct NodeTable(Mutex<Inner>);

impl NodeTable {
    pub fn new(root_fd: OwnedFd, root_key: FileKey) -> NodeTable {
        let mut nodes = HashMap::new();
        let mut by_key = HashMap::new();
        nodes.insert(ROOT_NODE, Node {
            fd: Arc::new(root_fd), key: root_key, generation: 0, nlookup: u64::MAX,
        });
        by_key.insert(root_key, ROOT_NODE);
        NodeTable(Mutex::new(Inner { nodes, by_key, next_id: ROOT_NODE + 1, next_generation: 1 }))
    }

    pub fn register(&self, fd: OwnedFd, key: FileKey) -> (NodeId, u64, Arc<OwnedFd>) {
        let mut g = self.0.lock().unwrap();
        if let Some(&id) = g.by_key.get(&key) {
            let node = g.nodes.get_mut(&id).unwrap();
            node.nlookup = node.nlookup.saturating_add(1);
            return (id, node.generation, Arc::clone(&node.fd));
            // `fd` drops here: the table already owns an fd for this file.
        }
        let id = g.next_id;
        g.next_id += 1;
        let generation = g.next_generation;
        g.next_generation += 1;
        let fd = Arc::new(fd);
        g.nodes.insert(id, Node { fd: Arc::clone(&fd), key, generation, nlookup: 1 });
        g.by_key.insert(key, id);
        (id, generation, fd)
    }

    pub fn get(&self, node: NodeId) -> Option<(Arc<OwnedFd>, u64)> {
        let g = self.0.lock().unwrap();
        g.nodes.get(&node).map(|n| (Arc::clone(&n.fd), n.generation))
    }

    pub fn forget(&self, node: NodeId, nlookup: u64) {
        if node == ROOT_NODE {
            return;
        }
        let mut g = self.0.lock().unwrap();
        let remove = match g.nodes.get_mut(&node) {
            Some(n) => {
                n.nlookup = n.nlookup.saturating_sub(nlookup);
                n.nlookup == 0
            }
            None => false,
        };
        if remove {
            if let Some(n) = g.nodes.remove(&node) {
                g.by_key.remove(&n.key);
            }
        }
    }
}
```

```rust
fn node_fd(&self, node: NodeId) -> FsResult<Arc<OwnedFd>> {
    self.nodes.get(node).map(|(fd, _)| fd).ok_or(Errno::ESTALE)

/// openat2 with the escape-proof resolve flags; O_PATH | O_NOFOLLOW so the
/// result can be a symlink node without dereferencing it.
async fn open_child(&self, parent: &OwnedFd, name: &CString) -> std::io::Result<OwnedFd> {

        .resolve(io_uring::types::ResolveFlags::BENEATH | io_uring::types::ResolveFlags::NO_MAGICLINKS);
```

```rust
// 1. HELLO: read_header + read_body; decode HelloRequest; check magic ==
//    MAGIC and version == PROTOCOL_VERSION (else reply status
//    STATUS_VERSION_MISMATCH and close). Settle limits:
//      max_inflight = min(client, cfg).clamp(WINDOW_CLAMP)
//      max_io_size  = min(client, cfg)
//    Reply HelloReply on STATUS_OK.
// 2. ATTACH: decode AttachRequest; allow.check(Path::new(OsStr::from_bytes(&path)))
//      Err(NotExported) -> STATUS_NOT_EXPORTED reply, close
//      Err(Denied)      -> STATUS_ATTACH_DENIED reply, close
//    Build LocalFs::new(&canonical, cfg.fsync, uring.clone(), pool.clone());
//    reply AttachReply { root_attr } (from fs.getattr(ROOT_NODE, None)).
// 3. Split the socket. Spawn the writer task:
//      mpsc::Receiver<OutFrame> -> write_frame(...)  (sole writer)
//    Reader loop:
//      read_header
//      window: sem.try_acquire_owned() -- failure means the client overran
//        the settled window => protocol violation => close connection
//      Opcode::try_from(hdr.op_or_status) -- unknown => violation => close
//      body = read_body(r, hdr.body_len, max_body)  -- oversize => close
//      data: only Write/Setxattr may carry it; others with data_len > 0 =>
//        violation. Write: data_len <= max_io_size, read into pool.get().
//        Setxattr: data_len <= max_body, read into Vec.
//      FORGET (flags & FLAG_NO_REPLY): run inline, no reply, release permit.
//      Everything else: tokio::spawn(handle_request(...)) owning the permit;
//        it calls dispatch() and sends OutFrame { request_id, status, body,
//        data } to the writer channel; permit drops after the send.
```

```rust
pub async fn dispatch(
    op: Opcode,
    body: &[u8],
    data: Option<DataIn>, // Pooled(PooledBuf) for Write, Owned(Vec<u8>) for Setxattr
    fs: &Arc<dyn FileSystem>,
) -> (u16, Vec<u8>, Option<DataPayload>) {
    match op {
        Opcode::Lookup => {
            let req: LookupRequest = match postcard::from_bytes(body) {
                Ok(r) => r,
                Err(_) => return (Errno::EINVAL.0, Vec::new(), None),
            };
            match fs.lookup(req.parent, &req.name).await {
                Ok(entry) => (STATUS_OK, postcard::to_allocvec(&entry).unwrap(), None),
                Err(e) => (e.0, Vec::new(), None),
            }
        }
        Opcode::Read => {
            let req: ReadRequest = /* same decode */;
            match fs.read(req.node, req.fh, req.offset, req.size).await {
                Ok(buf) => (STATUS_OK, Vec::new(), Some(DataPayload::Pooled(buf))),
                Err(e) => (e.0, Vec::new(), None),
            }
        }
        // ... every other opcode follows the identical decode/call/encode
        // shape; replies per the table in Task 3's comments (Entry, FileAttr,
        // (), WriteReply, ReaddirReply, XattrReply + data, ...).
        Opcode::Getxattr => {
            let req: GetxattrRequest = /* decode */;
            match fs.getxattr(req.node, &req.name, req.size).await {
                Ok((size, bytes)) => (
                    STATUS_OK,
                    postcard::to_allocvec(&XattrReply { size }).unwrap(),
                    Some(DataPayload::Owned(bytes)),
                ),
                Err(e) => (e.0, Vec::new(), None),
            }
        }
        Opcode::Hello | Opcode::Attach | Opcode::Forget => unreachable!("handled in session"),
    }
}
```

```rust
pub struct Connection {
    out_tx: tokio::sync::mpsc::Sender<Outbound>,
    pending: Arc<Mutex<HashMap<u64, oneshot::Sender<Reply>>>>,
    window: Arc<tokio::sync::Semaphore>,
    next_id: AtomicU64,
    dead: Arc<AtomicBool>,
    forget_tx: tokio::sync::mpsc::UnboundedSender<(NodeId, u64)>,
    pub limits: HelloReply,
}

struct Outbound { id: u64, op: u16, flags: u16, body: Vec<u8>, data: Vec<u8> }
struct Reply { status: u16, body: Vec<u8>, data: Vec<u8> }
```

- `connect`: perform hello/attach inline on the socket (sequential `write_frame`/`read_header`/`read_body`), then `split`; spawn **writer** (drain `out_tx`, `write_frame`), **reader** (loop `read_header` → `read_body` → read `data_len` bytes → complete `pending[id]`; on any error mark dead + drain pending with a synthesized `Reply { status: EIO }`), **forget batcher** (collect into `Vec`, flush on 64 or `tokio::time::interval(500ms)` tick by sending a `ForgetRequest` frame with `FLAG_NO_REPLY` through `out_tx`).
- `call`: check `dead` → `EIO`; take a window permit; id = `next_id.fetch_add(1)`; register oneshot; send `Outbound`; `rx.await`; map `status` (`0` → decode `Rep` from body, errno range → `Errno`, protocol range → `EIO` after `tracing::error!`); permit held until the reply or the drain.
- Getattr after attach: `connect` returns `AttachReply.root_attr` directly.

```rust
impl fuser::Filesystem for LbfsFuse {
    fn init(&mut self, _req: &fuser::Request<'_>, config: &mut fuser::KernelConfig) -> Result<(), libc::c_int> {
        let max_io = self.conn.limits.max_io_size;
        let _ = config.set_max_write(max_io);
        let _ = config.set_max_readahead(max_io);
        if self.writeback {
            let _ = config.add_capabilities(fuser::consts::FUSE_WRITEBACK_CACHE);
        }
        let _ = config.add_capabilities(fuser::consts::FUSE_DO_READDIRPLUS | fuser::consts::FUSE_READDIRPLUS_AUTO);
        Ok(())
    }

    fn lookup(&mut self, _req: &fuser::Request<'_>, parent: u64, name: &std::ffi::OsStr, reply: fuser::ReplyEntry) {
        let conn = Arc::clone(&self.conn);
        let ttl = self.ttl;
        let name = name.as_bytes().to_vec(); // std::os::unix::ffi::OsStrExt
        self.rt.spawn(async move {
            match conn.call::<_, Entry>(Opcode::Lookup, &LookupRequest { parent, name }, vec![]).await {
                Ok((e, _)) => reply.entry(&ttl, &to_fuse_attr(&e.attr), e.generation),
                Err(e) => reply.error(e.0 as i32),
            }
        });
    }
    // Every other callback follows this exact shape: capture params + owned
    // Reply, spawn on self.rt, one typed conn.call, map Ok -> reply.<kind>,
    // Err(errno) -> reply.error(errno.0 as i32).
}
```

`vm/up.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail
source "$(dirname "$0")/lib.sh"
KERNEL="${1:-}"

mkdir -p "$IMAGES"
[ -f "$BASE_IMG" ] || curl -fL "$UBUNTU_IMG_URL" -o "$BASE_IMG"

virsh --connect qemu:///system net-info "$NET_NAME" >/dev/null 2>&1 || {
  virsh --connect qemu:///system net-define "$VM_DIR/net.xml"
  virsh --connect qemu:///system net-autostart "$NET_NAME"
}
virsh --connect qemu:///system net-start "$NET_NAME" 2>/dev/null || true

KEY="$(cat "$SSH_PUBKEY")"
sed "s|__SSH_KEY__|$KEY|" "$VM_DIR/cloud-init/user-data.tmpl.yaml" > "$IMAGES/user-data.yaml"

boot_args=()
if [ -n "$KERNEL" ]; then
  k="$VM_DIR/kernels/$KERNEL"
  [ -f "$k/vmlinuz" ] || { echo "missing $k/vmlinuz" >&2; exit 1; }
  boot_args=(--boot "kernel=$k/vmlinuz,initrd=$k/initrd.img,kernel_args=root=LABEL=cloudimg-rootfs ro console=ttyS0")
fi

declare -A MACS=([lbfs-server]="$SERVER_MAC" [lbfs-client]="$CLIENT_MAC")
for vm in lbfs-server lbfs-client; do
  overlay="$IMAGES/$vm.qcow2"
  qemu-img create -f qcow2 -b "$BASE_IMG" -F qcow2 "$overlay" 20G
  virt-install --connect qemu:///system \
    --name "$vm" --memory 2048 --vcpus 2 \
    --disk "path=$overlay,format=qcow2" \
    --import --osinfo ubuntu24.04 \
    --network "network=$NET_NAME,mac=${MACS[$vm]}" \
    --cloud-init "user-data=$IMAGES/user-data.yaml" \
    --noautoconsole "${boot_args[@]}"
done

echo "waiting for ssh..."
for ip in "$SERVER_IP" "$CLIENT_IP"; do
  for _ in $(seq 1 60); do vm_ssh "$ip" true 2>/dev/null && break; sleep 5; done
  vm_ssh "$ip" 'echo "$(hostname): $(uname -r)"'
done
```

`vm/tests/e2e.sh` (runs **on the client VM**; server address as `$1`):

```bash
#!/usr/bin/env bash
set -euo pipefail
SERVER="${1:?server addr}"
MNT=/mnt/lbfs
EXPORT=/srv/exports/data

lbfs-client "$SERVER" "$EXPORT" "$MNT" &
CLIENT_PID=$!
trap 'kill $CLIENT_PID 2>/dev/null; fusermount3 -u $MNT 2>/dev/null || true' EXIT
for _ in $(seq 1 50); do grep -q " $MNT fuse" /proc/mounts && break; sleep 0.2; done
grep -q " $MNT fuse" /proc/mounts || { echo "FAIL: mount"; exit 1; }

cd "$MNT"
echo hello > f.txt;                 [ "$(cat f.txt)" = hello ]
mkdir -p d1/d2;                     mv f.txt d1/d2/
ln -s d1/d2/f.txt sym;              [ "$(cat sym)" = hello ]
ln d1/d2/f.txt hard;                [ "$(stat -c%h hard)" -eq 2 ]
truncate -s 3 hard;                 [ "$(stat -c%s d1/d2/f.txt)" -eq 3 ]
chmod 600 d1/d2/f.txt;              [ "$(stat -c%a d1/d2/f.txt)" = 600 ]
touch -d @1000000000 d1/d2/f.txt;   [ "$(stat -c%Y d1/d2/f.txt)" -eq 1000000000 ]
setfattr -n user.tag -v v1 d1/d2/f.txt
[ "$(getfattr --only-values -n user.tag d1/d2/f.txt)" = v1 ]
setfattr -x user.tag d1/d2/f.txt
dd if=/dev/zero of=sparse bs=1 count=1 seek=10M status=none
cp --sparse=auto sparse sparse2     # exercises lseek SEEK_DATA/SEEK_HOLE
cp d1/d2/f.txt copied               # exercises copy_file_range where cp uses it
find . -type f | sort > /tmp/listing.txt
grep -q "./copied" /tmp/listing.txt
rm -r d1 sym hard sparse sparse2 copied
[ -z "$(ls -A)" ]
cd /

# Integrity + throughput.
fio --directory="$MNT" /tmp/fio-verify.job
fio --name=seqread --directory="$MNT" --rw=read --bs=1M --size=512M \
    --ioengine=psync --direct=0 --numjobs=1 --group_reporting
echo "E2E OK"
```
